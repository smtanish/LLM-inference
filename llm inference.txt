#naive_decode.py
import torch
from transformers import AutoTokenizer,AutoModelForCausalLM
import time
import matplotlib.pyplot as plt 
model_name="Qwen/Qwen2.5-0.5B-Instruct"
tokenizer=AutoTokenizer.from_pretrained(model_name)
model=AutoModelForCausalLM.from_pretrained(
    model_name,
    dtype=torch.float16,
    device_map="cuda"
)
model.eval()
print("model loaded")
def naive_decode(prompt,max_new_tokens):
    messages=[
        {
            "role":"user",
            "content":prompt,
        }
    ]
    text=tokenizer.apply_chat_template(
        messages,
        tokenize=False,
        add_generation_prompt=True
    )
    inputs=tokenizer(
        text,
        return_tensors="pt"
    )
    input_ids=inputs["input_ids"].to(model.device)
    for _ in range(3):
        with torch.inference_mode():
            model(
                input_ids=input_ids,
                use_cache=False,
                logits_to_keep=1
            )
    generated_tokens=[]
    eos_token_ids=model.generation_config.eos_token_id
    if isinstance(eos_token_ids,int):
        eos_token_ids=[eos_token_ids]
    latencies=[]
    sequence_lengths=[]
    for step in range(max_new_tokens):
        torch.cuda.synchronize()
        t0=time.perf_counter()
        with torch.inference_mode():
            outputs=model(
                input_ids=input_ids,
                use_cache=False,
                logits_to_keep=1
            )
        torch.cuda.synchronize()
        t1=time.perf_counter()
        latencies.append((t1-t0)*1000)
        sequence_lengths.append(input_ids.shape[1])
        logits=outputs.logits[:,-1,:]
        next_token=torch.argmax(
            logits,
            dim=-1,
            keepdim=True
        )
        generated_tokens.append(next_token.item())
        if next_token.item() in eos_token_ids:
            break
        input_ids=torch.cat(
            [input_ids,next_token],
            dim=1
        )
    response=tokenizer.decode(
        generated_tokens,
        skip_special_tokens=True
    )
    print(response)
    return generated_tokens,latencies,sequence_lengths
prompt="explain machine learning in one sentence."
manual_tokens,manual_latencies,manual_sequence_lengths=naive_decode(prompt,50)
print("Latency measurements:",len(manual_latencies))
print("Sequence lengths:",len(manual_sequence_lengths))
plt.plot(manual_latencies)
plt.xlabel("decoding step")
plt.ylabel("latency(ms)")
plt.title("naive decoding: latency vs step")
plt.show()
print("step 0 latency: ",manual_latencies[0],"ms")
print("step 29 latency: ",manual_latencies[29],"ms")
print("step 29/step 0 ratio:",manual_latencies[29]/manual_latencies[0])
total_time=sum(manual_latencies)/1000
tokens_per_sec=len(manual_latencies)/total_time
print("total uncached time: ",total_time,"sec")
print("total uncached tokens/second: ",tokens_per_sec)

import torch
from transformers import AutoTokenizer,AutoModelForCausalLM
model_name="Qwen/Qwen2.5-0.5B-Instruct"
tokenizer=AutoTokenizer.from_pretrained(model_name)
model=AutoModelForCausalLM.from_pretrained(
    model_name,
    dtype=torch.float16,
    device_map="cuda"
)
model.eval()
def kv_prefill(prompt):
    messages=[
        {
            "role":"user",
            "content":prompt,
        }
    ]
    text=tokenizer.apply_chat_template(
        messages,
        tokenize=False,
        add_generation_prompt=True
    )
    inputs=tokenizer(
        text,
        return_tensors="pt"
    ).to(model.device)
    with torch.inference_mode():
        outputs=model(
            input_ids=inputs["input_ids"],
            use_cache=True,
            logits_to_keep=1
        )
    cache=outputs.past_key_values
    logits=outputs.logits[:,-1,:]

    next_token=torch.argmax(
        logits,
        dim=-1,
        keepdim=True
    )
    print("prompt length:",inputs["input_ids"].shape[1])
    print("cache length:",cache.get_seq_length())
    print("first token:",next_token.item())
    print("cache check:",cache.get_seq_length()==inputs["input_ids"].shape[1])
    return inputs["input_ids"],next_token,cache

prompt="explain machine learning in one sentence."
prefill_input_ids,next_token,cache=kv_prefill(prompt)
generated_tokens=[next_token.item()]

if isinstance(eos_token_ids,int):
    eos_token_ids=[eos_token_ids]

    current_position=cache.get_seq_length()
    position_ids=torch.tensor(
        [[current_position]],
        device=model.device
    )
    attention_mask=torch.ones(
        (1,current_position+1),
        dtype=torch.long,
        device=model.device
    )
    with torch.inference_mode():
        outputs=model(
            input_ids=next_token,
            past_key_values=cache,
            position_ids=position_ids,
            attention_mask=attention_mask,
            use_cache=True,
            logits_to_keep=1
        )
    next_token=torch.argmax(
        outputs.logits[:,-1,:],
        dim=-1,
        keepdim=True
    )
    generated_tokens.append(next_token.item())
    if next_token.item() in eos_token_ids:
        break
print("generated tokens:",generated_tokens)

#benchmark_kv.py
import torch
from transformers import AutoTokenizer, AutoModelForCausalLM
import time
import matplotlib.pyplot as plt
# load model
model_name = "Qwen/Qwen2.5-0.5B-Instruct"
tokenizer = AutoTokenizer.from_pretrained(model_name)
model = AutoModelForCausalLM.from_pretrained(
    model_name,
    dtype=torch.float16,
    device_map="cuda"
)
model.eval()
#prepare prompt
prompt = "explain machine learning in one sentence."
messages = [
    {
        "role": "user",
        "content": prompt
    }
]
text = tokenizer.apply_chat_template(
    messages,
    tokenize=False,
    add_generation_prompt=True
)
inputs = tokenizer(
    text,
    return_tensors="pt"
).to(model.device)
input_ids = inputs["input_ids"]
max_new_tokens = 30
#uncached decoder
def benchmark_uncached():
    current_ids = input_ids.clone()
    step_times = []
    # Warmup
    for _ in range(3):
        with torch.inference_mode():
            model(
                input_ids=current_ids,
                use_cache=False,
                logits_to_keep=1
            )
    torch.cuda.synchronize()
    for step in range(max_new_tokens):
        torch.cuda.synchronize()
        t0 = time.perf_counter()
        with torch.inference_mode():
            outputs = model(
                input_ids=current_ids,
                use_cache=False,
                logits_to_keep=1
            )
        torch.cuda.synchronize()
        t1 = time.perf_counter()
        step_times.append((t1 - t0) * 1000)
        logits = outputs.logits[:, -1, :]
        next_token = torch.argmax(
            logits,
            dim=-1,
            keepdim=True
        )
        current_ids = torch.cat(
            [current_ids, next_token],
            dim=1
        )
    return step_times
#kv cache decoder
def benchmark_cached():
    # Prefill
    with torch.inference_mode():
        outputs = model(
            input_ids=input_ids,
            use_cache=True,
            logits_to_keep=1
        )
    cache = outputs.past_key_values
    next_token = torch.argmax(
        outputs.logits[:, -1, :],
        dim=-1,
        keepdim=True
    )
    step_times = []
    # Warmup cached decode
    warmup_cache = cache
    for _ in range(3):
        current_position = warmup_cache.get_seq_length()
        position_ids = torch.tensor(
            [[current_position]],
            device=model.device
        )
        attention_mask = torch.ones(
            (1, current_position + 1),
            dtype=torch.long,
            device=model.device
        )
        with torch.inference_mode():
            model(
                input_ids=next_token,
                past_key_values=warmup_cache,
                position_ids=position_ids,
                attention_mask=attention_mask,
                use_cache=True,
                logits_to_keep=1
            )
    # Re-run prefill so timing starts with a fresh cache
    with torch.inference_mode():
        outputs = model(
            input_ids=input_ids,
            use_cache=True,
            logits_to_keep=1
        )
    cache = outputs.past_key_values
    next_token = torch.argmax(
        outputs.logits[:, -1, :],
        dim=-1,
        keepdim=True
    )
    torch.cuda.synchronize()
    for step in range(max_new_tokens - 1):
        current_position = cache.get_seq_length()
        position_ids = torch.tensor(
            [[current_position]],
            device=model.device
        )
        attention_mask = torch.ones(
            (1, current_position + 1),
            dtype=torch.long,
            device=model.device
        )
        torch.cuda.synchronize()
        t0 = time.perf_counter()
        with torch.inference_mode():
            outputs = model(
                input_ids=next_token,
                past_key_values=cache,
                position_ids=position_ids,
                attention_mask=attention_mask,
                use_cache=True,
                logits_to_keep=1
            )
        torch.cuda.synchronize()
        t1 = time.perf_counter()
        step_times.append((t1 - t0) * 1000)
        next_token = torch.argmax(
            outputs.logits[:, -1, :],
            dim=-1,
            keepdim=True
        )
    return step_times
#run benchmarks
uncached_times = benchmark_uncached()
cached_times = benchmark_cached()
#print results
print("\nUncached:")
print("Average:", sum(uncached_times) / len(uncached_times), "ms/token")
print("Total:", sum(uncached_times), "ms")
print("\nKV cached:")
print("Average:", sum(cached_times) / len(cached_times), "ms/token")
print("Total:", sum(cached_times), "ms")
speedup = (
    sum(uncached_times) / len(uncached_times)
) / (
    sum(cached_times) / len(cached_times)
)
print("\nAverage speedup:", speedup, "x")
#plot
plt.figure(figsize=(10, 6))
plt.plot(
    range(1, len(uncached_times) + 1),
    uncached_times,
    label="Uncached"
)
plt.plot(
    range(1, len(cached_times) + 1),
    cached_times,
    label="KV Cache"
)
plt.xlabel("Generation step")
plt.ylabel("Forward-pass latency (ms)")
plt.title("Uncached vs KV-Cache Decoding")
plt.legend()
plt.grid(True)
plt.tight_layout()
plt.show()
